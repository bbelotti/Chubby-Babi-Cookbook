\recipe{Béchamel sauce}{The final volume will be roughly equal to that of the milk.}{}{}
\label{sauce_bechamel}

\subsection*{Ingredients}
\begin{ingredients}
	\item 1 part flour
	\item 1 part butter
	\item 10 parts hot milk
	\item Grated nutmeg (to taste)
	\item Salt and pepper (to taste)
\end{ingredients}

\medskip

% --- Preparation ---
Melt the \textbf{butter} over low heat.
Add the \textbf{flour} and whisk vigorously to form a roux\footnote{Do not let the roux brown too much.}. 
Gradually pour in the \textbf{milk} while stirring constantly, and cook over medium heat until the sauce reaches the desired consistency. 

Off the heat, add \textbf{salt}, \textbf{pepper}, and \textbf{grated nutmeg} to taste.
