\recipe{Café de Paris}{A rich Swiss steak herbs sauce}{}{}

\begin{multicols}{2}
	\subsection*{Ingredients}
	\begin{ingredients}
		\item 125 g butter
		\item 1 egg yolk
		\item 1 generous tsp Dijon mustard
		\item \sfrac{1}{2} tbsp Cognac (or Brandy)
		\item \sfrac{1}{2} tsp white wine
		\item \sfrac{1}{2} tsp rosemary
		\item \sfrac{1}{2} tsp parsley
		\item 1 small anchovy
		\item \sfrac{1}{2} tsp thyme
		\item \sfrac{1}{2} tsp ground shallot
		\item \sfrac{1}{2} tsp ground garlic
		\item \sfrac{1}{2} tsp mild curry powder
		\item Pinch of salt
		\item Pinch of pepper
		\item Pinch of paprika
		
	\end{ingredients}
\end{multicols}

\medskip

% --- Preparation ---
Place the softened \textbf{butter} and \textbf{all the other ingredients} in a bowl.
Mash thoroughly and mix with a fork until a paste forms.
Using plastic wrap, shape into a log and let it firm up in the refrigerator. Melt a tbsp or two on your steak.
