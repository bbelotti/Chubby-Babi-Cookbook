\recipe{Jean's blondies}{Makes a 9x13 in (23x33 cm) pan.}{}{}

\subsection*{Ingredients}
\begin{ingredients}
	\item 2 \sfrac{1}{4} cups (280 g) flour
	\item 2 cups (400 g) brown sugar
	\item 1 tsp baking powder
	\item \sfrac{1}{2} tsp salt
	\item 1 cup (230 g) melted butter (unsalted)
	\item 2 eggs
	\item 1 tbsp (15 ml) vanilla extract
	\item Chocolate chips\footnote{Milk or white chocolate.} (to taste).
	\item Ice cream (optional).
\end{ingredients}

\medskip

% --- Preparation ---
In a large bowl, mix \textbf{all the dry ingredients}.
Mix in \textbf{all the wet ingredients} and the \textbf{chocolate chips}. 

Pour the batter in a greased 9x13 in (23x33 cm) baking pan and bake for 30-35 mins at 350°F (175°C). 
Eat warm with a scoop of ice cream on top.
